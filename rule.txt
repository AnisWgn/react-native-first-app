rules_version = '2';
service cloud.firestore {
 match /databases/{database}/documents {
    // Lecture autorisée aux utilisateurs connectés
 match /genres/{document} {
 allow read: if request.auth != null;
 // Écriture seulement si admin
 allow write: if request.auth != null
 && get(/databases/$(database)/documents/utilisateurs/$(request.auth.uid)).data.roles ==
"admin";
 }

  match /jeux/{document} {
 allow read: if request.auth != null;
 // Écriture seulement si admin
 allow write: if request.auth != null
 && get(/databases/$(database)/documents/utilisateurs/$(request.auth.uid)).data.roles ==
"admin";
 }

  match /marques/{document} {
 allow read: if request.auth != null;
 // Écriture seulement si admin
 allow write: if request.auth != null
 && get(/databases/$(database)/documents/utilisateurs/$(request.auth.uid)).data.roles ==
"admin";
 }

  match /pegis/{document} {
 allow read: if request.auth != null;
 // Écriture seulement si admin
 allow write: if request.auth != null
 && get(/databases/$(database)/documents/utilisateurs/$(request.auth.uid)).data.roles ==
"admin";
 }

  match /plateformes/{document} {
 allow read: if request.auth != null;
 // Écriture seulement si admin
 allow write: if request.auth != null
 && get(/databases/$(database)/documents/utilisateurs/$(request.auth.uid)).data.roles ==
"admin";
 }

 // Profils utilisateurs : chaque user lit les profils ; crée uniquement le sien (rôle user) ;
 // ne peut pas modifier son propre rôle (promotion admin = Console Firebase ou script admin).
 match /utilisateurs/{userId} {
   allow read: if request.auth != null;
   allow create: if request.auth != null
     && request.auth.uid == userId
     && request.resource.data.roles == 'user';
   allow update: if request.auth != null
     && request.auth.uid == userId
     && request.resource.data.roles == resource.data.roles;
   allow delete: if false;
 }
 }
}
