rules_version = '2';
service cloud.firestore {
 match /databases/{database}/documents {
    // Lecture autorisée aux utilisateurs connectés
 match /genres/{document} {
 allow read: if request.auth != null;
 // Écriture seulement si admin
 allow write: if request.auth != null
 && get(/databases/$(database)/documents/utilisateurs/$(request.auth.uid)).data.role ==
"admin";
 }

  match /jeux/{document} {
 allow read: if request.auth != null;
 // Écriture seulement si admin
 allow write: if request.auth != null
 && get(/databases/$(database)/documents/utilisateurs/$(request.auth.uid)).data.role ==
"admin";
 }

  match /marques/{document} {
 allow read: if request.auth != null;
 // Écriture seulement si admin
 allow write: if request.auth != null
 && get(/databases/$(database)/documents/utilisateurs/$(request.auth.uid)).data.role ==
"admin";
 }

  match /pegis/{document} {
 allow read: if request.auth != null;
 // Écriture seulement si admin
 allow write: if request.auth != null
 && get(/databases/$(database)/documents/utilisateurs/$(request.auth.uid)).data.role ==
"admin";
 }

  match /plateformes/{document} {
 allow read: if request.auth != null;
 // Écriture seulement si admin
 allow write: if request.auth != null
 && get(/databases/$(database)/documents/utilisateurs/$(request.auth.uid)).data.role ==
"admin";
 }

 // Collection users protégée
 match /utilisateurs/{userId} {
 allow read: if request.auth != null;
 allow write: if false;
 }
 }
}
