rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Función para verificar que el usuario logueado tenga el rol "admin"
    function isAdmin() {
      return request.auth != null && 
             get(/databases/$(database)/documents/utilisateurs/$(request.auth.uid)).data.role == 'admin';
    }
    // Permisos para las colecciones principales (leer para todos, escribir solo admins)
    match /genres/{document=**} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /jeux/{document=**} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /Plateformes/{document=**} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /Pegis/{document=**} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /Marques/{document=**} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /Activite/{document=**} {
    allow read: if true;
    allow write: if isAdmin();
    }
    
    // Regla para que los usuarios puedan leer su propio rol
    match /utilisateurs/{userId} {
      allow read: if request.auth != null && request.auth.uid == userId;
      // Solo tú desde la consola de Firebase puedes cambiar los roles manualmente,
      // esto evita que cualquier persona cambie su rol a admin.
      allow write: if false; 
    }
  }
}
